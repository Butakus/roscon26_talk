\begin{frame}{El camino clásico / recomendado}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{apt en Ubuntu}
\begin{itemize}
  \item El camino documentado y probado por la buildfarm.
  \item Cero compilación de ROS.
  \item Pero: una Ubuntu concreta por distro de ROS, y sin fijar versiones.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{alertblock}{Desde fuentes en tu sistema}
Ejemplo en Fedora 44:
\begin{itemize}
  \item 150 paquetes de ROS~2 (solo lo necesario).
  \item Múltiples dependencias de \texttt{dnf}; rosdep con reglas incompletas.
  \item Python 3.14, GCC 16, CMake 4.3: más nuevos que cualquier Tier 1.
\end{itemize}
\textbf{Posible, pero no recomendado.}
\end{alertblock}
\end{columns}
\end{frame}

%-------------------------------------------------
\begin{frame}[fragile]{Contenedores: Ubuntu, pero dentro de una caja}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Docker / Podman + devcontainer}
\begin{itemize}
  \item Ubuntu + ROS + \texttt{rosdep}, en una imagen.
  \item ROS desde apt o compilado desde fuentes.
  \item El código se monta desde el host.
  \item Windows y macOS, vía VM (Docker~Desktop).
\end{itemize}
\vspace{0.3em}
{\small Imagen 5,3~GB · build 2,5~min / 15~min}
\column{0.5\textwidth}
\textbf{Distrobox}
\begin{terminal}
distrobox assemble create
distrobox enter easynav-rolling
rosdep install --from-paths src -y
colcon build
\end{terminal}
\begin{itemize}
  \item Mismo usuario, \texttt{\$HOME}, red, GUI y USB.
  \item Solo Linux.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Pixi: un lockfile, tres sistemas operativos}
\begin{columns}[T]
\column{0.55\textwidth}
\begin{filelisting}
[workspace]
channels = ["robostack-rolling", "conda-forge"]
platforms = ["linux-64", "osx-arm64", "win-64"]

[dependencies]
ros-rolling-rclcpp = "*"

[tasks]
build = "colcon build"
\end{filelisting}
\begin{terminal}
pixi install && pixi run demo
\end{terminal}
\column{0.42\textwidth}
\begin{itemize}
  \item Sin root, sin tocar el sistema.
  \item ROS~2 como paquetes conda (RoboStack).
  \item \lstinline|pixi.lock| para las tres plataformas.
  \item 1,5~min de setup.
\end{itemize}
\vspace{0.3em}
\begin{mensaje}
{\small Workshop de Pixi + ROS~2:}\\
{\scriptsize\ttfamily\href{https://github.com/IntelligentRoboticsLabs/workshop-pixi}{github.com/IntelligentRoboticsLabs/\\workshop-pixi}\par}
\end{mensaje}
\end{columns}
\end{frame}

%-------------------------------------------------
\begin{frame}[fragile]{Conan: ROS~2 como una dependencia C++ más}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{filelisting}
[requires]
ros-kilted/2026.06.17

[options]
ros-kilted/*:variant=desktop
\end{filelisting}
\begin{itemize}
  \item ROS~2 en tu proyecto C++, igual que Boost u OpenCV.
  \item Sin \texttt{rosdep} ni \texttt{apt}, en Linux, macOS y Windows.
  \item Binarios compartibles en el servidor de la empresa.
\end{itemize}
{\small Instalación ${\sim}$15~min · build 2~min}
\column{0.46\textwidth}
\begin{block}{Un proyecto en maduración}
\begin{itemize}
  \item \texttt{ros-conan}: todo ROS~2 en \textbf{una única receta}, con variantes (\texttt{core}, \texttt{base}, \texttt{desktop}\ldots).
  \item Por ahora, solo \textbf{kilted}.
  \item Lo que no está en la receta, al workspace.
\end{itemize}
\end{block}
\vspace{0.5em}
% \begin{mensaje}
% Ideal si tu proyecto ya usa Conan.
% \end{mensaje}
\end{columns}
\end{frame}
